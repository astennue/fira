   677→-- Auto-create user profile on auth user creation
   678→CREATE OR REPLACE FUNCTION public.handle_new_user()
   679→RETURNS TRIGGER AS $$
   680→BEGIN
   681→  INSERT INTO public.users (id, email, display_name, first_name, last_name, role)
   682→  VALUES (
   683→    NEW.id,
   684→    NEW.email,
   685→    COALESCE(NEW.raw_user_meta_data->>'full_name', NEW.email),
   686→    COALESCE(NEW.raw_user_meta_data->>'full_name', split_part(NEW.email, '@', 1)),
   687→    '',
   688→    COALESCE(NEW.raw_user_meta_data->>'role', 'employee')
   689→  );
   690→  INSERT INTO public.public_profiles (id, display_name, first_name, last_name, role)
   691→  VALUES (
   692→    NEW.id,
   693→    COALESCE(NEW.raw_user_meta_data->>'full_name', NEW.email),
   694→    COALESCE(NEW.raw_user_meta_data->>'full_name', split_part(NEW.email, '@', 1)),
   695→    '',
   696→    COALESCE(NEW.raw_user_meta_data->>'role', 'employee')
   697→  );
   698→  RETURN NEW;
   699→END;
   700→$$ LANGUAGE plpgsql SECURITY DEFINER;

   703→DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
   704→CREATE TRIGGER on_auth_user_created
   705→  AFTER INSERT ON auth.users
   706→  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
