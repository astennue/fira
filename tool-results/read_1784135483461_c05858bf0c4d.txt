    92→// ── Password strength ────────────────────────────────────────
    93→function getPasswordStrength(password: string) {
    94→  if (!password) return { score: 0, label: '', color: '', width: '0%' };
    95→  let score = 0;
    96→  if (password.length >= 8) score++;
    97→  if (password.length >= 12) score++;
    98→  if (/[a-z]/.test(password) && /[A-Z]/.test(password)) score++;
    99→  if (/\d/.test(password)) score++;
   100→  if (/[^a-zA-Z\d]/.test(password)) score++;
   101→  switch (score) {
   102→    case 0:
   103→    case 1:
   104→      return { score, label: 'Weak', color: 'bg-red-500', width: '20%' };
   105→    case 2:
   106→      return { score, label: 'Fair', color: 'bg-yellow-500', width: '40%' };
   107→    case 3:
   108→      return { score, label: 'Good', color: 'bg-yellow-500', width: '60%' };
   109→    case 4:
   110→      return { score, label: 'Strong', color: 'bg-green-500', width: '80%' };
   111→    case 5:
   112→      return { score, label: 'Very Strong', color: 'bg-green-500', width: '100%' };
   113→    default:
   114→      return { score: 0, label: '', color: '', width: '0%' };
   115→  }
   116→}
   117→
   118→function PasswordStrengthMeter({ password }: { password: string }) {
   119→  const strength = getPasswordStrength(password);
   120→  if (!password) return null;
   121→  return (
   122→    <div className="space-y-1.5" role="status" aria-live="polite">
   123→      <div className="flex h-1.5 w-full overflow-hidden rounded-full bg-muted">
   124→        <motion.div
   125→          initial={{ width: 0 }}
   126→          animate={{ width: strength.width }}
   127→          transition={{ duration: 0.3, ease: 'easeOut' }}
   128→          className={`h-full rounded-full ${strength.color}`}
   129→        />
   130→      </div>
   131→      <p className="text-xs text-muted-foreground">
   132→        Password strength:{' '}
   133→        <span className={strength.score >= 3 ? 'font-medium text-green-600' : 'font-medium text-muted-foreground'}>
   134→          {strength.label}
   135→        </span>
   136→      </p>
   137→    </div>
   138→  );
   139→}
