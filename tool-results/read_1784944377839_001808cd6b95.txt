    46→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
    47→  const [count, setCount] = useState(0)
    48→  const ref = useRef<HTMLDivElement>(null)
    49→  const isInView = useInView(ref, { once: true })
    50→
    51→  useEffect(() => {
    52→    if (!isInView) return
    53→    let start = 0
    54→    const step = target / (duration / 16)
    55→    const timer = setInterval(() => {
    56→      start += step
    57→      if (start >= target) {
    58→        setCount(target)
    59→        clearInterval(timer)
    60→      } else {
    61→        setCount(Math.floor(start))
    62→      }
    63→    }, 16)
    64→    return () => clearInterval(timer)
    65→  }, [isInView, target, duration])
    66→
    67→  return <div ref={ref}>{count.toLocaleString()}{suffix}</div>
    68→}

    70→function TestimonialsCarousel({ testimonials }: { testimonials: any[] }) {
    71→  const [current, setCurrent] = useState(0)
    72→  const [isTransitioning, setIsTransitioning] = useState(false)
    73→
    74→  useEffect(() => {
    75→    if (testimonials.length <= 1) return
    76→    const timer = setInterval(() => {
    77→      setIsTransitioning(true)
    78→      setTimeout(() => {
    79→        setCurrent((prev) => (prev + 1) % testimonials.length)
    80→        setIsTransitioning(false)
    81→      }, 400)
    82→    }, 5000)
    83→    return () => clearInterval(timer)
    84→  }, [testimonials.length])
    85→
    86→  if (testimonials.length === 0) return null
    87→  const t = testimonials[current]
    88→
    89→  return (
    90→    <div className="relative min-h-[220px]">
    91→      <motion.div
    92→        key={t.id}
    93→        initial={{ opacity: 0, x: 20 }}
    94→        animate={{ opacity: isTransitioning ? 0 : 1, x: isTransitioning ? -20 : 0 }}
    95→        transition={{ duration: 0.4 }}
    96→        className="glass-card rounded-2xl p-6 md:p-8 max-w-2xl mx-auto text-center"
    97→      >
    98→        <div className="flex justify-center gap-1 mb-4">
    99→          {Array.from({ length: t.rating || 5 }).map((_, i) => (
   100→            <Star key={i} className="h-5 w-5 text-yellow-400 fill-yellow-400" />
   101→          ))}
   102→        </div>
   103→        <p className="text-white/90 text-base md:text-lg italic leading-relaxed mb-6">
   104→          &ldquo;{t.feedback}&rdquo;
   105→        </p>
   106→        <div className="flex items-center justify-center gap-3">
   107→          {t.avatar ? (
   108→            <img src={t.avatar} alt={t.name} className="h-10 w-10 rounded-full object-cover border-2 border-white/30" />
   109→          ) : (
   110→            <div className="h-10 w-10 rounded-full bg-white/20 flex items-center justify-center text-white font-bold">
   111→              {t.name?.charAt(0)}
   112→            </div>
   113→          )}
   114→          <div className="text-left">
   115→            <p className="text-white font-semibold text-sm">{t.name}</p>
   116→            {t.position && <p className="text-white/60 text-xs">{t.position}{t.company ? ` at ${t.company}` : ''}</p>}
   117→          </div>
   118→        </div>
   119→      </motion.div>
   120→      <div className="flex justify-center gap-2 mt-6">
   121→        {testimonials.map((_, i) => (
   122→          <button
   123→            key={i}
   124→            onClick={() => { setCurrent(i); setIsTransitioning(false) }}
   125→            className={`h-2 rounded-full transition-all duration-300 ${i === current ? 'w-8 bg-white' : 'w-2 bg-white/40 hover:bg-white/60'}`}
   126→          />
   127→        ))}
   128→      </div>
   129→    </div>
   130→  )
   131→}
