    69→/* ============================================================
    70→   ANIMATED COUNTER
    71→   ============================================================ */
    72→
    73→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
    74→  const [count, setCount] = useState(0)
    75→  const ref = useRef<HTMLDivElement>(null)
    76→  const isInView = useInView(ref, { once: true })
    77→
    78→  useEffect(() => {
    79→    if (!isInView) return
    80→    let start = 0
    81→    const step = target / (duration / 16)
    82→    const timer = setInterval(() => {
    83→      start += step
    84→      if (start >= target) {
    85→        setCount(target)
    86→        clearInterval(timer)
    87→      } else {
    88→        setCount(Math.floor(start))
    89→      }
    90→    }, 16)
    91→    return () => clearInterval(timer)
    92→  }, [isInView, target, duration])
    93→
    94→  return <div ref={ref}>{count.toLocaleString()}{suffix}</div>
    95→}

   101→function TestimonialsCarousel({ testimonials }: { testimonials: any[] }) {
   102→  const [current, setCurrent] = useState(0)
   103→  const [direction, setDirection] = useState(0)
   104→
   105→  const paginate = (dir: number) => {
   106→    setDirection(dir)
   107→    setCurrent((prev) => (prev + dir + testimonials.length) % testimonials.length)
   108→  }
   109→
   110→  useEffect(() => {
   111→    if (testimonials.length <= 1) return
   112→    const timer = setInterval(() => paginate(1), 6000)
   113→    return () => clearInterval(timer)
   114→  }, [testimonials.length])
   115→
   116→  if (testimonials.length === 0) return null
   117→
   118→  const variants = {
   119→    enter: (dir: number) => ({ x: dir > 0 ? 200 : -200, opacity: 0 }),
   120→    center: { x: 0, opacity: 1 },
   121→    exit: (dir: number) => ({ x: dir < 0 ? 200 : -200, opacity: 0 }),
   122→  }
   123→
   124→  const t = testimonials[current]
   125→
   126→  return (
   127→    <div className="relative">
   128→      <div className="overflow-hidden">
   129→        <AnimatePresence initial={false} custom={direction} mode="wait">
   130→          <motion.div
   131→            key={current}
   132→            custom={direction}
   133→            variants={variants}
   134→            initial="enter"
   135→            animate="center"
   136→            exit="exit"
   137→            transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
   138→            className="glass-card rounded-2xl p-8 md:p-10 max-w-2xl mx-auto text-center"
   139→          >
   140→            <Quote className="h-8 w-8 text-blue-400/60 mx-auto mb-4" />
   141→            <div className="flex justify-center gap-1 mb-5">
   142→              {Array.from({ length: t.rating || 5 }).map((_, i) => (
   143→                <Star key={i} className="h-4 w-4 text-amber-400 fill-amber-400" />
   144→              ))}
   145→            </div>
   146→            <p className="text-foreground/90 text-base md:text-lg italic leading-relaxed mb-6">
   147→              &ldquo;{t.feedback}&rdquo;
   148→            </p>
   149→            <div className="flex items-center justify-center gap-3">
   150→              {t.avatar ? (
   151→                <img src={t.avatar} alt={t.name} className="h-11 w-11 rounded-full object-cover border-2 border-blue-400/40" />
   152→              ) : (
   153→                <div className="h-11 w-11 rounded-full bg-blue-500/20 flex items-center justify-center text-foreground font-bold text-sm">
   154→                  {t.name?.charAt(0)}
   155→                </div>
   156→              )}
   157→              <div className="text-left">
   158→                <p className="text-foreground font-semibold text-sm">{t.name}</p>
   159→                {t.position && <p className="text-muted-foreground text-xs">{t.position}{t.company ? ` at ${t.company}` : ''}</p>}
   160→              </div>
   161→            </div>
   162→          </motion.div>
   163→        </AnimatePresence>
   164→      </div>
   165→      {/* Navigation */}
   166→      <div className="flex items-center justify-center gap-4 mt-8">
   167→        <button onClick={() => paginate(-1)} className="h-9 w-9 rounded-full glass flex items-center justify-center hover:bg-blue-500/20 transition-colors">
   168→          <ChevronLeft className="h-4 w-4" />
   169→        </button>
   170→        <div className="flex gap-2">
   171→          {testimonials.map((_, i) => (
   172→            <button
   173→              key={i}
   174→              onClick={() => { setDirection(i > current ? 1 : -1); setCurrent(i) }}
   175→              className={`h-2 rounded-full transition-all duration-500 ${i === current ? 'w-8 bg-blue-500' : 'w-2 bg-muted-foreground/30 hover:bg-muted-foreground/50'}`}
   176→            />
   177→          ))}
   178→        </div>
   179→        <button onClick={() => paginate(1)} className="h-9 w-9 rounded-full glass flex items-center justify-center hover:bg-blue-500/20 transition-colors">
   180→          <ChevronRight className="h-4 w-4" />
   181→        </button>
   182→      </div>
   183→    </div>
   184→  )
   185→}
