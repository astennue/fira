    55→// Safe date parser handling string | Date | Firestore Timestamp-like shapes.
    56→function toDate(v: any): Date | null {
    57→  if (!v) return null;
    58→  if (v instanceof Date) return v;
    59→  if (typeof v === "string" || typeof v === "number") {
    60→    const d = new Date(v);
    61→    return isNaN(d.getTime()) ? null : d;
    62→  }
    63→  if (v?.toDate) {
    64→    try { return v.toDate(); } catch { return null; }
    65→  }
    66→  if (v?.seconds != null) return new Date(v.seconds * 1000);
    67→  return null;
    68→}
